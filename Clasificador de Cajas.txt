  <h3>🔍 Bajo la solapa: la lógica del clasificador</h3>
  <p>Esto que ves moverse no es magia: es un <b>bucle de control</b> que se repite miles de veces por segundo en toda fábrica real.</p>
  <h4>El principio físico</h4>
  <p>El sensor mide la altura <i>h</i> de cada caja. La decisión del PLC es una simple comparación:</p>
  <div class="formula">h<sub>caja</sub> &gt; h<sub>umbral</sub> &nbsp;⇒&nbsp; activar brazo B1</div>
  <p>Y para que el brazo no llegue tarde, el PLC calcula el <b>retardo de sincronización</b>: como la caja viaja a velocidad <i>v</i> y hay una distancia <i>d</i> entre el sensor y el brazo:</p>
  <div class="formula">t<sub>espera</sub> = d<sub>S1→B1</sub> / v<sub>banda</sub></div>
  <p>Con <i>d = 6 u</i> y <i>v = 1.6 u/s</i>, el PLC espera <b>3.75 s</b>… pero en una fábrica real, con <i>d = 0.9 m</i> y <i>v = 0.3 m/s</i>, serían <b>3 s</b>. Cambia la velocidad en el panel y observa cómo el pulso luminoso viaja justo a tiempo.</p>
  <h4>Código real: diagrama escalera (ladder) del PLC</h4>
  <pre>|  S1(h &gt; TH)        B1_SOLICITA
|——] [——] [———————————( )——|

|  B1_SOLICITA  TON[T#t_esp]
|——] [—————————————(TON)——|  // espera d/v

|  TON.DN                   B1
|——] [—————————————————————( )——|  // extiende brazo

/* Pseudocódigo equivalente */
IF sensor.altura > umbral:
    esperar(d_sensor_brazo / v_banda)
    brazo_B1.extender(0.5 s)
    contador_desviadas += 1</pre>

const lblArm = makeLabel('B1 · brazo',0xa33327,.7); lblArm.position.set(ARM_X,2.6,-1.2); arm.add(lblArm);
scene.add(arm);
let armT = 0; // 0 retracted → 1 extended

/* ============ Signal lines (glowing paths) ============ */
function makeFlow(points, color){
  const g = new THREE.BufferGeometry().setFromPoints(points.map(p=>new THREE.Vector3(...p)));
  const m = new THREE.LineDashedMaterial({color, dashSize:.45, gapSize:.25,
    transparent:true, opacity:0.15});
  const line = new THREE.Line(g,m); line.computeLineDistances(); scene.add(line);
  return {line, mat:m};
}
const flowS1 = makeFlow([[SENSOR_X,3,1],[SENSOR_X,3.6,-4],[-2,3.4,-6]], 0x2ec4b6); // sensor→PLC
const flowPLC = makeFlow([[-1,-6],[ARM_X,-3.5],[ARM_X,1.5,-2.2]], 0xf4a53f);     // PLC→arm
const flowEnd = makeFlow([[END_X,0,0],[END_X+6,0,0],[END_X+8,-2,2]], 0x9a8f7a);  // belt→out
function pulseFlow(f, strength){ f.mat.opacity = Math.min(.95, f.mat.opacity + strength); }

/* ============ Boxes ============ */
const boxes = [];
const boxGeo = new THREE.BoxGeometry(1.6,1,1.6); // unit height scaled
const nextHeights = [];
function spawnBox(height, ghost=false){
  const mat = new THREE.MeshStandardMaterial({
    color: ghost?0x2ec4b6:0xf7f1e3, roughness:.85,
    transparent: ghost, opacity: ghost?.35:1 });
  const m = new THREE.Mesh(boxGeo, mat);
  m.scale.y = height;
  m.position.set(-BELT_LEN/2-1.5, height/2-.17, 0);
  m.castShadow = true;
  const e = inkEdges(boxGeo,.8); e.scale.y=height; m.add(e);
  scene.add(m);
  const b = {mesh:m, h:height, measured:false, divert:false, pushed:false, falling:false, ghost};
  boxes.push(b);
  return b;
}
function removeBox(b){ scene.remove(b.mesh); b.mesh.material.dispose();
  b.mesh.children[0].geometry.dispose(); boxes.splice(boxes.indexOf(b),1); }

/* ============ Simulation state ============ */
const S = {
  v: 1.6, threshold: 1.2,
  total:0, div:0, pass:0, predOK:0,
  predicting:false, ghost:null, pendingH:null,
  stamp:false
};
const randH = ()=> .6 + Math.random()*1.4;

/* ============ Prediction flow ============ */
function enterPredict(){
  if(S.predicting) return;
  S.predicting = true; S.pendingH = randH();
  S.ghost = spawnBox(S.pendingH, true);
  $('pH').textContent = S.pendingH.toFixed(2);
  $('pT').textContent = S.threshold.toFixed(2);
  $('predictbox').classList.add('show');
  setState('Prediciendo… la caja fantasma espera tu veredicto','predict');
  toast('🔮 Modo predicción: la caja fantasma espera tu decisión');
}
function resolvePrediction(said){
  if(!S.predicting) return;
  const willDiv = S.pendingH > S.threshold;
  const ok = (said==='div') === willDiv;
  S.predicting = false;
  $('predictbox').classList.remove('show');
  if(S.ghost){ removeBox(S.ghost); S.ghost=null; }
  const b = spawnBox(S.pendingH); b.prediction = ok ? 1 : -1;
  S.pendingH = null;
  if(ok){ S.predOK++; toast('🎯 ¡Predicción correcta! La física está de tu lado.');
    checkStamp(); }
  else toast('❌ Predicción fallida: revisa el umbral y vuelve a intentarlo.');
  setState('Probando tu predicción en la línea…');
  updateHUD();
}

/* ============ Stamp ============ */

  if(S.stamp) return;
  if(S.predOK>=3 && S.total>=10){

    const st = $('stamp'); st.classList.add('earned');
    st.innerHTML = '✦ ING.<br>LÍNEA ✦';
    toast('🏅 ¡Sello ganado: Ingeniero de Línea!');
    setState('Sello ganado. ¡Lección dominada!','predict');
  }
}

/* ============ Animation loop ============ */
const clock = new THREE.Clock();
const HOLD = (ARM_X - SENSOR_X); // distance sensor→arm
function animate(){
  requestAnimationFrame(animate);
  const dt = Math.min(clock.getDelta(), .05);
  const t = clock.elapsedTime;

  // belt dashes scroll
  dashes.forEach(d=>{ d.position.x += S.v*dt; if(d.position.x>BELT_LEN/2) d.position.x -= BELT_LEN; });

  // idle signal shimmer
  flowS1.mat.opacity = .12 + .06*Math.sin(t*2);
  flowPLC.mat.opacity = .10 + .05*Math.sin(t*2+1);
  lamp.material.color.setHex(Math.sin(t*3)>0?0x2ec4b6:0x555555);
  plcLamp.material.color.setHex(Math.sin(t*2.4)>0?0xf4a53f:0x555555);

  // arm extension easing
  armT += ((arm.active?1:0)-armT)*Math.min(1,dt*7);
  rod.position.z = -1.4 + armT*1.6;
  plate.position.z = -.15 + armT*1.6;
  arm.active = false;

  for(let i=boxes.length-1;i>=0;i--){
    const b = boxes[i];
    if(b.ghost){ // ghost hover at spawn
      b.mesh.position.y = b.h/2-.17 + .12*Math.sin(t*3);
      b.mesh.rotation.y = Math.sin(t)*0.06;
      continue;
    }
    if(!b.falling){
      b.mesh.position.x += S.v*dt;
      // measure at sensor
      if(!b.measured && b.mesh.position.x >= SENSOR_X){
        b.measured = true;
        pulseFlow(flowS1, .8);
        beamMat.opacity = .75;
        b.divert = b.h > S.threshold;
        b.reachArmAt = SENSOR_X + HOLD; // sync: constant distance
      }
      if(b.measured && b.divert && !b.pushed && b.mesh.position.x >= b.reachArmAt){
        pulseFlow(flowPLC, .9);
        arm.active = true;
        b.pushed = true;
      }
      if(b.pushed && b.mesh.position.z < 4.2){
        b.mesh.position.z += 3.2*dt; // pushed sideways
        b.mesh.position.x = ARM_X;
        if(b.mesh.position.z >= 4.2){
          S.div++; S.total++; pulseFlow(flowPLC,.6);
          if(b.prediction===1) checkStamp();
          toast(b.prediction===1?'🎯 ¡Correcto! → desviada':
                b.prediction===-1?'❌ Predijiste mal: fue desviada':'📦 Caja alta → desviada');
          removeBox(b); updateHUD(); continue;
        }
      }
      if(b.mesh.position.x > END_X){
        b.falling = true; pulseFlow(flowEnd,.5);
      }
    } else {
      b.mesh.position.x += S.v*dt*.6;
      b.mesh.position.y -= 3.5*dt;
      b.mesh.rotation.z -= dt*.8;
      if(b.mesh.position.y < -3){
        S.pass++; S.total++;
        toast(b.prediction===1?'❌ Fue desviada… ¡no! Continuó.':
              b.prediction===-1?'🎯 ¡Correcto! continuó':'📦 Caja baja → sigue la banda');
        removeBox(b); updateHUD(); checkStamp(); continue;
      }
    }
  }

  // decay effects
  beamMat.opacity = Math.max(.16, beamMat.opacity - dt*1.6);
  [flowS1,flowPLC,flowEnd].forEach(f=> f.mat.opacity = Math.max(f.mat.opacity - dt*.7, .1));

  controls.update();
  renderer.render(scene,camera);
}
updateHUD();
animate();

  renderer.setSize(innerWidth, innerHeight);
